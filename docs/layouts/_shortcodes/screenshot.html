{{- /* Theme-aware screenshot from assets/images/-light.png and -dark.png (the PNGs are the source of truth, taken at 2x; WebP is generated here so no image tool is needed outside Hugo). The screenshot links to its full-size image, which works without JavaScript and keeps open-in-new-tab. assets/js/screenshot.js turns a plain click into a modal lightbox with the image of the active theme; the link targets the light image because the theme is unknown without JavaScript. priority="true" marks the screenshot as the page's LCP image (see _partials/fm/screenshot-picture.html). Usage: {{< screenshot name="web-ui" alt="…" class="fm-ribbon" priority="true" >}} */ -}} {{- $name := .Get "name" -}} {{- $alt := .Get "alt" | default "" -}} {{- $class := .Get "class" | default "" -}} {{- $img := dict -}} {{- range $scheme := slice "light" "dark" -}} {{- $path := printf "images/%s-%s.png" $name $scheme -}} {{- with resources.Get $path -}} {{- $img = merge $img (dict $scheme (.Process "webp q85")) -}} {{- else -}} {{- errorf "screenshot: assets/%s not found (%s)" $path $.Position -}} {{- end -}} {{- end -}} {{- $id := printf "fm-shot-%s-%d" $name .Ordinal -}} {{- $picture := dict "light" $img.light "dark" $img.dark "alt" $alt -}}
{{- partial "fm/screenshot-picture.html" $picture -}} {{- if not (.Page.Store.Get "fm-lightbox-script") -}} {{- .Page.Store.Set "fm-lightbox-script" true -}} {{- with resources.Get "js/screenshot.js" | fingerprint }} {{- end -}} {{- end -}}
